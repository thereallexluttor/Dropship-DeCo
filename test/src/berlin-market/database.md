create table public.pagos_pendientes (
  id serial not null,
  request_id character varying(255) not null,
  pedido_id integer not null,
  referencia character varying(255) null,
  monto numeric(10, 2) not null,
  estado character varying(50) null default 'PENDING'::character varying,
  ultima_verificacion timestamp with time zone null default now(),
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  email_comprador text null,
  nombre_comprador text null,
  telefono_comprador text null,
  constraint pagos_pendientes_pkey primary key (id),
  constraint pagos_pendientes_request_id_key unique (request_id),
  constraint fk_pedido foreign KEY (pedido_id) references pedidos (id) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_pagos_pendientes_request_id on public.pagos_pendientes using btree (request_id) TABLESPACE pg_default;

create index IF not exists idx_pagos_pendientes_pedido_id on public.pagos_pendientes using btree (pedido_id) TABLESPACE pg_default;

create index IF not exists idx_pagos_pendientes_estado on public.pagos_pendientes using btree (estado) TABLESPACE pg_default;

create index IF not exists idx_pagos_pendientes_ultima_verificacion on public.pagos_pendientes using btree (ultima_verificacion) TABLESPACE pg_default;

create index IF not exists idx_pagos_pendientes_estado_verificacion on public.pagos_pendientes using btree (estado, ultima_verificacion) TABLESPACE pg_default
where
  (
    (estado)::text = any (
      (
        array[
          'PENDING'::character varying,
          'PENDING_VALIDATION'::character varying
        ]
      )::text[]
    )
  );

create trigger trigger_update_pagos_pendientes_updated_at BEFORE
update on pagos_pendientes for EACH row
execute FUNCTION update_pagos_pendientes_updated_at ();
